<For each={threads.data()}>{(t) => <li>{t.title}</li>}</For>`;

export const reactCode = `import { useDb, useQuery } from '@spooky-sync/client-react';

const db = useDb<typeof schema>();
const threads = useQuery(
  db.query('thread').orderBy('created_at', 'desc').limit(20).build()
);

if (threads.isLoading) return <Spinner />;
return threads.data.map((t) => <li key={String(t.id)}>{t.title}</li>);`;

export const svelteCode = `<script lang="ts">
  import { useDb, useQuery } from '@spooky-sync/client-svelte';

  const db = useDb<typeof schema>();
  const threads = useQuery(() =>
    db.query('thread').orderBy('created_at', 'desc').limit(20).build()
  );
</script>

{#each threads.data as t (String(t.id))}<li>{t.title}</li>{/each}`;

export const flutterCode = `// db is the AppDb generated from your schema
final Stream<List<Thread>> threads = db.thread
    .query()
    .orderBy(Thread$.createdAt, desc: true)
    .limit(20)
    .watch();

StreamBuilder<List<Thread>>(
  stream: threads,
  builder: (context, snap) => ListView(
    children: [for (final t in snap.data ?? const <Thread>[]) Text(t.title)],
  ),
)`;

export const rustCode = `let mut threads = client
    .live(client.query("thread").order_by("created_at", "desc").limit(20))
    .await?;

println!("{} threads from the local store", threads.rows().len());

// Every change after: a local write, another device, the web app.
while let Ok(rows) = threads.changed().await {
    println!("{} threads", rows.len());
}`;

export const vanillaCode = `import { client } from './db';

const { hash } = await client
  .query('thread', {})
  .orderBy('created_at', 'desc')
  .limit(20)
  .build()
  .run();

const unsubscribe = await client.subscribe(hash, renderThreads, { immediate: true });`;

sp00ky has one sync engine and a client for each place your app runs. The TypeScript core
(`@spooky-sync/core`) runs in the browser, and the Solid, Solid 2, React and Svelte bindings wrap it in
each framework's own reactivity. Flutter apps use the Dart client, and desktop apps, CLIs and
services use the Rust client. They all talk to the same server with the same protocol, so a web
app, a phone and a desktop app can share one database and see each other's writes live.

| Client | Install | Runs on | Guide |
| --- | --- | --- | --- |
| SolidJS | `@spooky-sync/client-solid` | Browser, Solid 1.x | [SolidJS guide](/docs/guide/solid) |
| Solid 2 (RC) | `@spooky-sync/client-solid2` | Browser, Solid 2.0 RC | [Solid 2 guide](/docs/guide/solid2) |
| React | `@spooky-sync/client-react` | Browser, React 18 and 19 | [React guide](/docs/guide/react) |
| Svelte | `@spooky-sync/client-svelte` | Browser, Svelte 5.29+ (runes) | [Svelte guide](/docs/guide/svelte) |
| Flutter | `spooky_core` and `spooky_flutter` (Dart, from git) | Flutter apps on iOS and Android | [Flutter guide](/docs/guide/flutter) |
| Rust | `spooky-core` crate (from git) | Desktop apps, CLIs and services | [Rust guide](/docs/guide/rust) |
| Vanilla JS / TS | `@spooky-sync/core` | Browser, with any framework or none | [Vanilla guide](/docs/guide/vanilla) |

<Note title="React Native">
  The TypeScript clients run SurrealDB and SQLite as WebAssembly with IndexedDB or OPFS and web
  workers, which React Native does not have. For a mobile app, use the [Flutter](/docs/guide/flutter)
  client.
</Note>

## How each one hands you data

| Client | Live results | Writes | Loading state |
| --- | --- | --- | --- |
| SolidJS | Accessors: `threads.data()` | `db.create` / `db.update` / `db.delete` | `isLoading()` |
| Solid 2 | Accessors, plus `ready()` for `<Loading>` suspension | Same, plus `createSubmission` for button state | `isLoading()`, `isEmpty()`, `isSettled()` |
| React | Plain values through `useSyncExternalStore`, rows keep their identity between updates | Same, plus `useSubmission` for button state | `isLoading`, `isEmpty`, `isSettled` |
| Svelte | Reactive getters on runes: `threads.data`, rows keep their identity between updates | Same, plus `useSubmission` for button state | `isLoading`, `isEmpty`, `isSettled` |
| Flutter | `Stream<List<T>>` from the generated `AppDb`, for a `StreamBuilder` | `create` / `update` / `delete` on the client or the typed tables | `isQueryAuthoritative`, `isQuerySettled` |
| Rust | `LiveQuery`: `rows()` now, `changed().await` for each update | `create` / `update` / `delete`, typed structs from `build.rs` | `is_authoritative()` and `status()` on the live query |
| Vanilla | `client.subscribe(hash, callback)` | `client.create` / `update` / `delete` | `subscribeQueryAuthority`, `subscribeQueryStatus` |

Every client writes to the local store first and syncs in the background, so writes work offline
and the UI updates before the server answers.

## One schema, every client

All clients read the same `.surql` schema. `spky generate` writes the TypeScript and Dart types
from the `clientTypes` entries in `sp00ky.yml`, and Rust apps generate theirs from a `build.rs`.
See [Generated types](/docs/schema/generated-types).
