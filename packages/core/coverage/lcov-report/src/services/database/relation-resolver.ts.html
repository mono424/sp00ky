import type { RelationPlan } from '@spooky-sync/query-builder';
import {
  MAX_RELATION_DEPTH,
  RelationCycleError,
  type OrderBy,
  type Row,
  type RowFetcher,
} from './cache-engine';
 
/**
 * Resolve the `.related()` tree of a query by **level-ordered, batched
 * fan-out** — the engine-neutral replacement for SurrealQL's nested
 * `(SELECT..) AS alias` projections. One batched fetch per relation per level
 * (never per-row N+1, never a flat join); grouping, per-parent ORDER/LIMIT and
 * attachment happen in JS.
 *
 * Correlation (mirrors `buildSubquery`):
 * - `one`  → parent[`foreignKeyField`] = child.id  → attach `bucket[0] ?? null`
 * - `many` → child[`foreignKeyField`] = parent.id   → attach `bucket`
 *
 * Cross-level dependency is handled by ordering: a child's correlation key is a
 * field on its parent, known only once the parent level has resolved, so each
 * level's batch runs after the previous level's rows exist. `parents` is
 * mutated in place — each resolved alias is appended LAST so key order matches
 * SurrealQL's `SELECT *, <sub> AS alias` (base fields, then alias).
 *
 * @throws {RelationCycleError} if nesting exceeds {@link MAX_RELATION_DEPTH}.
 */
export async function resolveRelations(
  parents: Row[],
  relations: RelationPlan[] | undefined,
  fetcher: RowFetcher,
  depth = 0,
  path: string[] = []
): Promise<void> {
  if (!relations || relations.length === 0 || parents.length === 0) return;
  if (depth >= MAX_RELATION_DEPTH) {
    throw new RelationCycleError([...path, relations.map((r) => r.alias).join('|')]);
  }
 
  // Siblings are independent — resolve them concurrently.
  await Promise.all(
    relations.map((relation) => resolveOne(parents, relation, fetcher, depth, path))
  );
}
 
async function resolveOne(
  parents: Row[],
  relation: RelationPlan,
  fetcher: RowFetcher,
  depth: number,
  path: string[]
): Promise<void> {
  const isOne = relation.cardinality === 'one';
  // The child field to match parent keys against.
  const matchField = isOne ? 'id' : relation.foreignKeyField;
  // How to read a parent's correlation key.
  const parentKeyOf = (parent: Row): unknown => (isOne ? parent[relation.foreignKeyField] : parent['id']);
 
  // Distinct, defined correlation keys — a null/absent FK contributes nothing
  // (never a spurious `WHERE id = null` fetch). For a to-one relation the key
  // must also LOOK like a record id: a `string | record<t>` column (whitepawn's
  // `game.white`, a legacy plain player name in old rows) carries values that
  // are not links, and `type::record(<string> "Magnus")` in the batched fetch
  // would throw and take every sibling's name down with it.
  const keySet = new Map<string, unknown>();
  for (const parent of parents) {
    const key = parentKeyOf(parent);
    if (key == null) continue;
    if (isOne && !looksLikeRecordId(key)) continue;
    const k = stableKey(key);
    if (!keySet.has(k)) keySet.set(k, key);
  }
 
  const grouped = new Map<string, Row[]>();
  if (keySet.size > 0) {
    const children = await fetcher.fetchRelation({
      table: relation.table,
      matchField,
      keys: [...keySet.values()],
      where: relation.where,
      orderBy: relation.orderBy,
      select: relation.select,
    });
 
    // Recurse into the next level BEFORE grouping — each child is a parent for
    // its own relations. Resolving the full deduped child set once (rather than
    // per parent) keeps nested fan-out at O(depth) batches. Children later
    // dropped by a per-parent LIMIT carry resolved nested data harmlessly (they
    // are never attached to any output row).
    await resolveRelations(children, relation.relations, fetcher, depth + 1, [
      ...path,
      relation.alias,
    ]);
 
    for (const child of children) {
      const k = stableKey(child[matchField]);
      const bucket = grouped.get(k);
      if (bucket) bucket.push(child);
      else grouped.set(k, [child]);
    }
  }
 
  for (const parent of parents) {
    const key = parentKeyOf(parent);
    let bucket = key == null ? [] : grouped.get(stableKey(key)) ?? [];
 
    // Per-parent ORDER then LIMIT — the batch fetch's order is only a hint; a
    // shared batch cannot express "top N per parent", so re-apply here.
    if (relation.orderBy && relation.orderBy.length > 0) {
      bucket = sortRows(bucket, relation.orderBy);
    }
    if (relation.limit !== undefined) {
      bucket = bucket.slice(0, relation.limit);
    }
 
    if (isOne) {
      // A to-one alias usually IS the foreign-key column (`.related('white')`
      // projects onto `white`). A dangling LINK resolves to null, as SurrealQL's
      // `(SELECT …)[0] AS alias` would. A value that is not a link at all (a
      // legacy plain player name in a `string | record<t>` column) is kept: it
      // was never going to resolve, and a renderer's fallback still needs it.
      const raw = parent[relation.alias];
      const keepRaw =
        relation.alias === relation.foreignKeyField && raw != null && !looksLikeRecordId(raw);
      parent[relation.alias] = bucket[0] ?? (keepRaw ? raw : null);
    } else {
      parent[relation.alias] = bucket;
    }
  }
}
 
/**
 * Whether a correlation value can address a record: a RecordId (duck-typed) or
 * a `table:id` string. Anything else (a plain display string, a number) cannot
 * be the target of a to-one relation and is skipped rather than fetched.
 */
export function looksLikeRecordId(value: unknown): boolean {
  if (value == null) return false;
  if (typeof value === 'string') return /^[A-Za-z_][A-Za-z0-9_]*:.+/.test(value);
  if (typeof value === 'object') {
    const rid = value as { tb?: unknown; id?: unknown };
    return rid.tb !== undefined && rid.id !== undefined;
  }
  return false;
}
 
/**
 * Stable grouping key for a correlation value. A `RecordId` (duck-typed to
 * avoid a hard surrealdb dependency here) and its `table:id` string form must
 * collapse to the same key so a parent FK stored either way matches the child.
 */
export function stableKey(value: unknown): string {
  if (value == null) return 'null';
  if (typeof value === 'string') return value;
  if (typeof value === 'number' || typeof value === 'boolean' || typeof value === 'bigint') {
    return String(value);
  }
  // Duck-type RecordId: { tb, id } or a toString yielding `table:id`.
  if (typeof value === 'object') {
    const rid = value as { tb?: unknown; id?: unknown; toString?: () => string };
    if (rid.tb !== undefined && rid.id !== undefined) {
      return `${String(rid.tb)}:${stableKey(rid.id)}`;
    }
    if (typeof rid.toString === 'function') {
      const s = rid.toString();
      if (s !== '[object Object]') return s;
    }
    return JSON.stringify(value);
  }
  return String(value);
}
 
/**
 * Stable, multi-key sort matching SurrealQL `ORDER BY` semantics closely
 * enough for cache display order: ascending null-last, type-aware compare, and
 * a stable tiebreak (original index) so equal rows keep input order.
 */
export function sortRows(rows: Row[], orderBy: OrderBy): Row[] {
  const indexed = rows.map((row, i) => [row, i] as const);
  indexed.sort((a, b) => {
    for (const [field, direction] of orderBy) {
      const cmp = compareValues(a[0][field], b[0][field]);
      if (cmp !== 0) return direction === 'desc' ? -cmp : cmp;
    }
    return a[1] - b[1]; // stable tiebreak
  });
  return indexed.map(([row]) => row);
}
 
function compareValues(a: unknown, b: unknown): number {
  if (a == null && b == null) return 0;
  if (a == null) return 1; // nulls sort last (ascending)
  if (b == null) return -1;
  if (typeof a === 'number' && typeof b === 'number') return a - b;
  const as = stableKey(a);
  const bs = stableKey(b);
  return as < bs ? -1 : as > bs ? 1 : 0;
}
